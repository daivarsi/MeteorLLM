cmake_minimum_required(VERSION 3.16)
project(MeteorLLMValidation CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Threads REQUIRED)
add_library(meteor_engine ../engine.cpp ../quant.cpp)
target_link_libraries(meteor_engine PUBLIC Threads::Threads)
add_executable(host-runner ../host_main.cpp)
target_link_libraries(host-runner PRIVATE meteor_engine)
add_executable(tokenizer-cli ../tokenizer.cpp ../tokenizer_cli.cpp)
add_executable(quant-cli ../quant.cpp ../quant_cli.cpp)
if(MSVC)
 add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
 target_compile_options(meteor_engine PRIVATE /O2 /fp:precise)
endif()
enable_testing()
add_test(NAME half_precision_roundtrip COMMAND host-runner --half-test)
